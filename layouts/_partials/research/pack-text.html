{{- /* Keep Markdown prose; turn rendered figure HTML into portable text. Links retain their destinations, including figure citations. */ -}} {{- $text := . -}} {{- $text = replaceRE `(?s)]*>.*?` "" $text -}} {{- $text = replaceRE `(?s)]*href="([^"]+)"[^>]*>(.*?)` "[$2]($1)" $text -}} {{- $text = replaceRE `(?s)]*>(.*?)` "\n\n### $1\n\n" $text -}} {{- $text = replaceRE `]*>` "\n- " $text -}} {{- $text = replaceRE `` "\n" $text -}} {{- $text = replaceRE `` " " $text -}} {{- $text = replaceRE `<(?:p|div|section|header|figcaption|figure)\b[^>]*>` "\n" $text -}} {{- $text = replaceRE `<[^>]+>` "" $text -}} {{- $text = htmlUnescape $text -}} {{- /* Root-relative destinations must survive outside the website. */ -}} {{- $text = replaceRE `\]\(/` "](https://gautsch.org/" $text -}} {{- $text = replaceRE `\n[ \t]+` "\n" $text -}} {{- $text = replaceRE `\n{3,}` "\n\n" $text -}} {{- return (strings.TrimSpace $text) -}}
